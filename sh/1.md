## 分享链接创建
### 1. 设置端口和uuid
SUB_PORT=$(shuf -i 20000-60000 -n 1)
SUB_UUID=$(cat /proc/sys/kernel/random/uuid)

mkdir -p /var/www/pxy-sub
# 直接复制纯文本（也可改为 base64 -w0 /etc/pxy/nodes.txt > /var/www/pxy-sub/$SUB_UUID 生成通用 Base64 订阅）
grep -v '^#' /etc/pxy/nodes.txt | grep -v '^$' > "/var/www/pxy-sub/$SUB_UUID"
chmod 755 /var/www/pxy-sub
chmod 644 "/var/www/pxy-sub/$SUB_UUID"

### 2. 配置独立的 HTTPS 订阅分享端口
cat > /etc/nginx/conf.d/pxy-sub.conf <<EOF
server {
    listen $SUB_PORT ssl http2;
    listen [::]:$SUB_PORT ssl http2;
    server_name api.touzi.dpdns.org;

    ssl_certificate     /etc/pxy/cert.pem;
    ssl_certificate_key /etc/pxy/cert.key;
    ssl_protocols       TLSv1.3 TLSv1.2;
    server_tokens       off;

    root /var/www/pxy-sub;
    autoindex off;

    # 仅允许精确匹配该 UUID 路径，其他任何探测请求直接断开连接
    location = /$SUB_UUID {
        default_type text/plain;
        charset utf-8;
        try_files \$uri =404;
    }

    location / {
        return 444;
    }
}
EOF

nginx -t && systemctl reload nginx

### 3. 放行防火墙端口并打印完整分享链接
if command -v ufw >/dev/null && ufw status | grep -q "Status: active"; then
  ufw allow "$SUB_PORT/tcp" >/dev/null
fi

echo "你的加密分享链接为："
echo "https://api.touzi.dpdns.org:$SUB_PORT/$SUB_UUID"

### 4. 配置证书续签后自动重载该端口
由于该新端口直接加载了 /etc/pxy/cert.pem，脚本原有的 /usr/local/sbin/pxy-cert-deploy 钩子中已经包含 systemctl reload nginx，因此后续 certbot 自动续签证书时，该分享端口也会自动同步更新证书，无需额外维护。   (注：如果之后修改了 /etc/pxy/nodes.txt 的内容，只需重新执行一次 grep -v '^#' /etc/pxy/nodes.txt | grep -v '^$' > "/var/www/pxy-sub/$SUB_UUID" 即可同步更新在线内容。)
